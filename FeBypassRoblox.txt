-- Execute Function (修復載入後的 GUI、支援多重拖曳與按鈕點擊)
executeButton.MouseButton1Click:Connect(function()
    local scriptText = inputBox.Text
    local assetId = scriptText:match("require%s*%(?%s*(%d+)%s*%)?") or scriptText:match("(%d+)")

    if assetId then
        print("[Executor] 嘗試載入require ID:", assetId)
        
        task.spawn(function()
            local success, result = pcall(function()
                return game:GetObjects("rbxassetid://" .. assetId)
            end)
            
            if success and result and result[1] then
                local obj = result[1]
                
                local targetGui = obj
                if not obj:IsA("ScreenGui") then
                    targetGui = Instance.new("ScreenGui")
                    targetGui.Name = "LoadedAssetContainer"
                    targetGui.DisplayOrder = 9999
                    targetGui.ResetOnSpawn = false
                    obj.Parent = targetGui
                    targetGui.Parent = playerGui
                else
                    obj.DisplayOrder = 9999
                    obj.Parent = playerGui
                end

                print("[Executor] 載入成功！正在全面修復遠端介面的拖曳與按鈕功能...")

                -- 核心修復：撕掉背景保鮮膜、讓所有主視窗皆可拖曳、解鎖內建按鈕
                for _, descendant in ipairs(targetGui:GetDescendants()) do
                    if descendant:IsA("Frame") then
                        local sizeX = descendant.AbsoluteSize.X
                        local screenX = workspace.CurrentCamera.ViewportSize.X
                        
                        -- 如果是全螢幕大背景則關閉 Active（保鮮膜防護），小的子視窗則開啟拖曳
                        if sizeX >= (screenX * 0.5) then
                            descendant.Active = false
                        else
                            descendant.Active = true
                            descendant.Draggable = true
                        end
                    elseif descendant:IsA("TextButton") or descendant:IsA("ImageButton") then
                        descendant.Active = true
                        descendant.Selectable = true
                        descendant.ZIndex = 50
                        
                        -- 嘗試強行喚醒內建按鈕的點擊監聽
                        pcall(function()
                            descendant.MouseButton1Click:Connect(function()
                                -- 觸發按鈕原生的事件或回調
                            end)
                        end)
                    elseif descendant:IsA("LocalScript") then
                        descendant.Disabled = false
                    end
                end

                -- 觸發模組腳本（如 .dominant("c00lkidd_Clan21") 的載入邏輯）
                for _, descendant in ipairs(targetGui:GetDescendants()) do
                    if descendant:IsA("ModuleScript") then
                        pcall(function()
                            require(descendant)
                        end)
                    end
                end

                print("[Executor] 遠端介面修復完畢，請測試 Inject、Execute 與 Clear！")
            else
                warn("[Executor] 載入失敗或伺服器拒絕存取該 Asset ID。")
            end
        end)
    else
        warn("[Executor] 無效的 require() 格式或 ID。")
    end
end)

        task.wait()

        local dirX = math.cos(Angle)
